local InputAlvos = MainTab:Input({
    Title = "PromptR's target",
    Desc = "alvo do painel",
    Callback = function(OpcaoSelecionada)
        if OpcaoSelecionada == "" then 
            AlvoSelecionado = ""
            return 
        end
        
        local argumento = string.lower(OpcaoSelecionada)
        local Players = game:GetService("Players")
        local LocalPlayer = Players.LocalPlayer

        local jogadorEncontrado = nil
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                -- Usando string.find para localizar o alvo por qualquer parte do Nome ou DisplayName
                if string.find(string.lower(p.Name), argumento) or string.find(string.lower(p.DisplayName), argumento) then
                    jogadorEncontrado = p.Name
                    break
                end
            end
        end

        if jogadorEncontrado then
            AlvoSelecionado = jogadorEncontrado
        else
            AlvoSelecionado = OpcaoSelecionada -- Mantém o texto bruto para o caso de strings globais ou carregamentos tardios
        end
    end
})

MainTab:Button({
    Title = ":punish",
    Desc = "Teleporta os cones para o alvo; se ele sentar, a cadeira vai para o limbo",
    Callback = function()
        if AlvoSelecionado ~= "" then
            local Players = game:GetService("Players")
            local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
            
            local targetPlayer = nil
            for _, player in pairs(Players:GetPlayers()) do
                if string.find(string.lower(player.Name), string.lower(AlvoSelecionado)) or string.find(string.lower(player.DisplayName), string.lower(AlvoSelecionado)) then
                    targetPlayer = player
                    break
                end
            end
            
            if targetPlayer then
                local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
                
                if trafficCones then
                    WindUI:Notify({
                        Title = "🔨 Punição Iniciada", 
                        Content = "Teleportando props. Se o alvo " .. targetPlayer.Name .. " sentar, a cadeira irá para o limbo.", 
                        Duration = 3
                    })

                    if _G.PunishAtivo then 
                        _G.PunishAtivo = false 
                        task.wait(0.1) 
                    end
                    _G.PunishAtivo = true

                    task.spawn(function()
                        while _G.PunishAtivo and targetPlayer and targetPlayer.Parent == Players and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") do
                            local character = targetPlayer.Character
                            local targetHRP = character.HumanoidRootPart
                            local humanoid = character:FindFirstChildOfClass("Humanoid")
                            
                            if humanoid and humanoid.Health <= 0 then
                                break
                            end
                            
                            -- Padrão caso o alvo não tenha perna (R6/R15 Fallback)
                            local destinoPunishCFrame = targetHRP.CFrame
                            
                            if humanoid and (humanoid.Sit or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat") or humanoid.SeatPart) then
                                destinoPunishCFrame = CFrame.new(99999999999999, -501, -9999999999999999)
                                
                                local cadeiraFisica = humanoid.SeatPart or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat")
                                if cadeiraFisica and cadeiraFisica:IsA("BasePart") then
                                    pcall(function()
                                        if cadeiraFisica.Anchored then
                                            cadeiraFisica.Anchored = false
                                        end
                                    end)
                                end
                            else
                                -- Se NÃO estiver sentado, puxa a perna direita (Compatível com R6 e R15)
                                local pernaDireita = character:FindFirstChild("RightLeg") 
                                    or character:FindFirstChild("RightLowerLeg") 
                                    or character:FindFirstChild("Right Leg")
                                
                                if pernaDireita and pernaDireita:IsA("BasePart") then
                                    destinoPunishCFrame = pernaDireita.CFrame
                                end
                            end
                            
                            local props = trafficCones:GetChildren()
                            
                            if #props > 0 then
                                for _, prop in pairs(props) do
                                    if not _G.PunishAtivo or not (targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")) then break end
                                    
                                    local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                                    if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                        task.spawn(function()
                                            pcall(function()
                                                setCFrameRemote:InvokeServer(destinoPunishCFrame)
                                            end)
                                        end)
                                    end
                                end
                            end
                            task.wait(0.1) 
                        end
                        
                        if _G.PunishAtivo then
                            WindUI:Notify({
                                Title = "✅ Punição Concluída", 
                                Content = "Punição encerrada com sucesso.", 
                                Duration = 5
                            })
                            _G.PunishAtivo = false
                        end
                    end)
                else
                    WindUI:Notify({Title = "❌ Erro", Content = "Pasta 001_TrafficCones não encontrada.", Duration = 3})
                end
            else
                WindUI:Notify({Title = "❌ Erro", Content = "Jogador não encontrado.", Duration = 3})
            end
        else
            WindUI:Notify({Title = "⚠️ Aviso", Content = "Por favor, selecione um alvo válido antes de usar.", Duration = 3})
        end
    end
})

MainTab:Button({
    Title = ":bring",
    Desc = "Teleporta os cones para a perna; se sentar, traz a cadeira até você, limpa os props e para",
    Callback = function()
        if AlvoSelecionado ~= "" then
            local Players = game:GetService("Players")
            local localPlayer = Players.LocalPlayer
            local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
            local ReplicatedStorage = game:GetService("ReplicatedStorage")
            
            local targetPlayer = nil
            for _, player in pairs(Players:GetPlayers()) do
                if string.find(string.lower(player.Name), string.lower(AlvoSelecionado)) or string.find(string.lower(player.DisplayName), string.lower(AlvoSelecionado)) then
                    targetPlayer = player
                    break
                end
            end
            
            if targetPlayer then
                local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
                
                if trafficCones then
                    WindUI:Notify({
                        Title = "🔨 Bring Iniciado", 
                        Content = "Monitorando perna de " .. targetPlayer.Name .. ". O bring ocorrerá quando ele sentar.", 
                        Duration = 3
                    })

                    task.spawn(function()
                        local finalizado = false

                        -- Loop focado no alvo até ele sentar
                        while targetPlayer and targetPlayer.Parent == Players and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") do
                            if finalizado then break end

                            local character = targetPlayer.Character
                            local targetHRP = character.HumanoidRootPart
                            local humanoid = character:FindFirstChildOfClass("Humanoid")
                            
                            if humanoid and humanoid.Health <= 0 then
                                break
                            end
                            
                            -- Fallback padrão caso a perna suma temporariamente
                            local destinoPunishCFrame = targetHRP.CFrame
                            local alvoSentou = false
                            
                            -- SE O ALVO SENTAR: Teleporta a cadeira para a sua posição (LocalPlayer)
                            if humanoid and (humanoid.Sit or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat") or humanoid.SeatPart) then
                                if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") then
                                    -- Posiciona um pouco à frente do seu personagem para não bugar dentro de você
                                    destinoPunishCFrame = localPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
                                end
                                alvoSentou = true
                                
                                local cadeiraFisica = humanoid.SeatPart or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat")
                                if cadeiraFisica and cadeiraFisica:IsA("BasePart") then
                                    pcall(function()
                                        if cadeiraFisica.Anchored then
                                            cadeiraFisica.Anchored = false
                                        end
                                    end)
                                end
                            else
                                -- Se NÃO estiver sentado, segue estritamente a perna direita (R6 / R15)
                                local pernaDireita = character:FindFirstChild("RightLeg") 
                                    or character:FindFirstChild("RightLowerLeg") 
                                    or character:FindFirstChild("Right Leg")
                                
                                if pernaDireita and pernaDireita:IsA("BasePart") then
                                    destinoPunishCFrame = pernaDireita.CFrame
                                end
                            end
                            
                            local props = trafficCones:GetChildren()
                            
                            if #props > 0 then
                                for _, prop in pairs(props) do
                                    if not (targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")) then break end
                                    
                                    local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                                    if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                        task.spawn(function()
                                            pcall(function()
                                                setCFrameRemote:InvokeServer(destinoPunishCFrame)
                                            end)
                                        end)
                                    end
                                end
                            end
                            
                            -- Se ele sentou nesta repetição, finaliza o loop e limpa os props
                            if alvoSentou then
                                finalizado = true
                                task.wait(0.1) -- Delay sutil para garantir que a cadeira chegue até você antes de sumir com os cones
                                
                                pcall(function()
                                    local reFolder = ReplicatedStorage:FindFirstChild("RE")
                                    local clearEvent = reFolder and reFolder:FindFirstChild("1Clea1rTool1s")
                                    if clearEvent then
                                        clearEvent:FireServer("ClearAllProps")
                                    end
                                end)
                                break 
                            end

-- SISTEMA DE COMANDOS VIA CHAT PARA ;SHUTDOWN (COLE NO FINAL DO ARQUIVO)
-- ====================================================================
local Players = game:GetService("Players")
local localPlayer = Players.LocalPlayer
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local function executarShutdownPorChat()
    local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local rootpart = character:WaitForChild("HumanoidRootPart", 10)
    if not rootpart then return end
    local oldcf = rootpart.CFrame

    local re = ReplicatedStorage:FindFirstChild("RE")
    local toolRemote = re and re:FindFirstChild("1Too1l")
    if not toolRemote then return end

    -- Notificação visual na WindUI do LocalPlayer
    pcall(function()
        WindUI:Notify({ 
            Title = "Shutdown", 
            Content = "O servidor esta sendo desligado e desconectando Players", 
            Duration = 5 
        })
    end)

    task.spawn(function()
        for m = 1, 500 do
            task.spawn(function()
                if toolRemote:IsA("RemoteFunction") then
                    toolRemote:InvokeServer("PickingTools", "FireHose")
                else
                    toolRemote:FireServer("PickingTools", "FireHose")
                end
            end)

            local backpack = localPlayer:FindFirstChild("Backpack")
            if backpack then
                local fireHose = backpack:FindFirstChild("FireHose")
                if fireHose and fireHose:FindFirstChild("ToolSound") then
                    fireHose.ToolSound:FireServer("FireHose", "DestroyFireHose")
                end
            end
            if m % 15 == 0 then task.wait(0.4) end
        end
    end)

    task.wait(0.4)
    localPlayer.CharacterRemoving:Wait()
    local newCharacter = localPlayer.CharacterAdded:Wait()
    local newRootPart = newCharacter:WaitForChild("HumanoidRootPart", 15)
    local humanoid = newCharacter:WaitForChild("Humanoid", 15)

    if newRootPart and humanoid then
        task.wait(0.7)
        humanoid:ChangeState(Enum.HumanoidStateType.Physics)
        newRootPart.CFrame = oldcf
        newRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    end

    task.wait(0.5)

    local backpack = localPlayer:WaitForChild("Backpack", 10)
    if backpack then
        local items = backpack:GetChildren()
        local equipCount = 0
        for i = 1, #items do
            local item = items[i]
            if item:IsA("Tool") and item.Name == "FireHose" then
                equipCount = equipCount + 1
                task.defer(function() item.Parent = newCharacter end)
                if equipCount % 8 == 0 then task.wait(0.02) end 
            end
        end
    end
    task.wait(0.5)
end

-- Escuta as mensagens enviadas no chat do jogo
local function aoFalarShutdown(message)
    -- Verifica se você digitou exatamente ";shutdown"
    if string.lower(message) == ":shut-down" then
        executarShutdownPorChat()
    end
end

-- Conecta o evento de chat garantindo a trava de segurança apenas para o seu usuário
if localPlayer and localPlayer:IsA("Player") then
    localPlayer.Chatted:Connect(aoFalarShutdown)
end

local function executarJumpPorChat(nomeDigitado)
    if nomeDigitado == "" then return end
    
    local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
    
    -- Encontra o alvo baseado no início do Username ou DisplayName (== 1)
    local targetPlayer = nil
    for _, player in pairs(Players:GetPlayers()) do
        if string.find(string.lower(player.Name), string.lower(nomeDigitado)) == 1 or string.find(string.lower(player.DisplayName), string.lower(nomeDigitado)) == 1 then
            targetPlayer = player
            break
        end
    end
    
    if targetPlayer then
        local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
        
        if trafficCones then
            if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
                local character = targetPlayer.Character
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                
                if humanoid and humanoid.Health <= 0 then
                    return
                end
                
                -- Fallback padrão caso a perna suma temporariamente
                local destinoCFrame = character.HumanoidRootPart.CFrame
                
                -- Segue estritamente a perna direita (Suporta R6 e R15)
                local pernaDireita = character:FindFirstChild("RightLeg") 
                    or character:FindFirstChild("RightLowerLeg") 
                    or character:FindFirstChild("Right Leg")
                
                if pernaDireita and pernaDireita:IsA("BasePart") then
                    destinoCFrame = pernaDireita.CFrame
                end
                
                local props = trafficCones:GetChildren()
                
                if #props > 0 then
                    -- PASSO 1: Teleporta os props (trampolim) para a perna direita do alvo
                    for _, prop in pairs(props) do
                        local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                        if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                            task.spawn(function()
                                pcall(function()
                                    setCFrameRemote:InvokeServer(destinoCFrame)
                                end)
                            end)
                        end
                    end
                    
                    -- PASSO 2: Aguarda o tempo exato (0.05s) para o motor do jogo registrar o encaixe
                    task.wait(0.5)
                    
                    -- PASSO 3: Apaga o Prop via RemoteEvent do servidor para forçar o pulo mecânico
                    pcall(function()
                        local reFolder = ReplicatedStorage:FindFirstChild("RE")
                        local clearEvent = reFolder and reFolder:FindFirstChild("1Clea1rTool1s")
                        if clearEvent then
                            clearEvent:FireServer("ClearAllProps")
                        end
                    end)
                    
                    -- Notificação na tela do LocalPlayer
                    pcall(function()
                        WindUI:Notify({
                            Title = "🛡️ Moderação (Chat)", 
                            Content = "Comando ;jump executado em " .. targetPlayer.Name, 
                            Duration = 2
                        })
                    end)
                end
            end
        end
    end
end

local function executarBringPorChat(nomeDigitado)
    if nomeDigitado == "" then return end
    
    local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
    
    -- Encontra o alvo baseado no início do Username ou DisplayName (== 1)
    local targetPlayer = nil
    for _, player in pairs(Players:GetPlayers()) do
        if string.find(string.lower(player.Name), string.lower(nomeDigitado)) == 1 or string.find(string.lower(player.DisplayName), string.lower(nomeDigitado)) == 1 then
            targetPlayer = player
            break
        end
    end
    
    if targetPlayer then
        local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
        
        if trafficCones then
            -- Dispara a notificação visual na WindUI do LocalPlayer
            pcall(function()
                WindUI:Notify({
                    Title = "🔨 Bring Iniciado (Chat)", 
                    Content = "Monitorando perna de " .. targetPlayer.Name .. ". O bring ocorrerá quando ele sentar.", 
                    Duration = 3
                })
            end)

            task.spawn(function()
                local finalizado = false

                -- Loop focado no alvo até ele sentar
                while targetPlayer and targetPlayer.Parent == Players and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") do
                    if finalizado then break end

                    local character = targetPlayer.Character
                    local targetHRP = character.HumanoidRootPart
                    local humanoid = character:FindFirstChildOfClass("Humanoid")
                    
                    if humanoid and humanoid.Health <= 0 then
                        break
                    end
                    
                    -- Fallback padrão caso a perna suma temporariamente
                    local destinoPunishCFrame = targetHRP.CFrame
                    local alvoSentou = false
                    
                    -- SE O ALVO SENTAR: Teleporta a cadeira para a sua posição (LocalPlayer)
                    if humanoid and (humanoid.Sit or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat") or humanoid.SeatPart) then
                        if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            -- Posiciona um pouco à frente do seu personagem para não travar um dentro do outro
                            destinoPunishCFrame = localPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
                        end
                        alvoSentou = true
                        
                        local cadeiraFisica = humanoid.SeatPart or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat")
                        if cadeiraFisica and cadeiraFisica:IsA("BasePart") then
                            pcall(function()
                                if cadeiraFisica.Anchored then
                                    cadeiraFisica.Anchored = false
                                end
                            end)
                        end
                    else
                        -- Se NÃO estiver sentado, segue estritamente a perna direita (R6 / R15)
                        local pernaDireita = character:FindFirstChild("RightLeg") 
                            or character:FindFirstChild("RightLowerLeg") 
                            or character:FindFirstChild("Right Leg")
                        
                        if pernaDireita and pernaDireita:IsA("BasePart") then
                            destinoPunishCFrame = pernaDireita.CFrame
                        end
                    end
                    
                    local props = trafficCones:GetChildren()
                    
                    if #props > 0 then
                        for _, prop in pairs(props) do
                            if not (targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")) then break end
                            
                            local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                            if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                task.spawn(function()
                                    pcall(function()
                                        setCFrameRemote:InvokeServer(destinoPunishCFrame)
                                    end)
                                end)
                            end
                        end
                    end
                    
                    if alvoSentou then
                        finalizado = true
                        task.wait(0.1) -- Garante que a cadeira replique até você antes de sumir com os cones
                        
                        pcall(function()
                            local reFolder = ReplicatedStorage:FindFirstChild("RE")
                            local clearEvent = reFolder and reFolder:FindFirstChild("1Clea1rTool1s")
                            if clearEvent then
                                clearEvent:FireServer("ClearAllProps")
                            end
                        end)
                        break 
                    end
                    
                    task.wait(0.1)
                end
                
                pcall(function()
                    WindUI:Notify({
                        Title = "✅ Comando Executado (Chat)", 
                        Content = "Alvo trazido e props limpos.", 
                        Duration = 5
                    })
                end)
            end)

-- Função de busca inteligente por iniciais ou parte do nome
local function encontrarJogadorPorIniciais(texto)
    local textoMinusculo = string.lower(texto)
    for _, jogador in pairs(Players:GetPlayers()) do
        local nickMinusculo = string.lower(jogador.Name)
        local displayMinusculo = string.lower(jogador.DisplayName)
        
        -- Verifica se o texto digitado bate com o início do Nick ou Display Name
        if string.sub(nickMinusculo, 1, #textoMinusculo) == textoMinusculo or 
           string.sub(displayMinusculo, 1, #textoMinusculo) == textoMinusculo then
            return jogador
        end
    end
    return nil
end
