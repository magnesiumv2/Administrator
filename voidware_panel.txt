-- 1. Carrega a versão mais recente e oficial da WindUI do repositório de releases
local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/latest/download/main.lua"))()

-- 2. Cria a Janela Principal (Padrão VoidWare)
local Window = WindUI:CreateWindow({
    Title = "anonymus panel",
    SubTitle = "Administração & Utilitários",
    Icon = "door-open",
    Size = UDim2.fromOffset(550, 380),
    Theme = "Dark"
})

local MainTab = Window:Tab({
    Title = "Commandos",
    Icon = "lucide-swords"
})

-- Variável global para armazenar o alvo selecionado
local AlvoSelecionado = ""

-- Função para obter a lista de jogadores atuais (excluindo o próprio jogador)
local function obterListaJogadores()
    local lista = {}
    for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
        if p ~= game:GetService("Players").LocalPlayer then
            table.insert(lista, p.Name)
        end
    end
    return lista
end

-- ============================================================
-- 1. SELEÇÃO DE ALVO (Dropdown Nativo da WindUI)
-- ============================================================
local DropdownAlvos = MainTab:Dropdown({
    Title = "Selecionar Alvo",
    Desc = "Escolha um jogador do servidor para executar ações",
    Values = obterListaJogadores(),
    Value = "",
    Callback = function(OpcaoSelecionada)
        if type(OpcaoSelecionada) == "table" then
            AlvoSelecionado = OpcaoSelecionada[1] or ""
        else
            AlvoSelecionado = OpcaoSelecionada or ""
        end
        print("Alvo atual definido para: " .. tostring(AlvoSelecionado))
    end
})

-- Sincronização automatizada da lista do dropdown
local function atualizarDropdown()
    local novaLista = obterListaJogadores()
    DropdownAlvos:Refresh(novaLista)
end

game:GetService("Players").PlayerAdded:Connect(function()
    task.wait(0.5)
    atualizarDropdown()
end)

game:GetService("Players").PlayerRemoving:Connect(function(player)
    if AlvoSelecionado == player.Name then
        AlvoSelecionado = ""
    end
    atualizarDropdown()
end)

                    if Ball and Ball:IsA("BasePart") then
                        Ball.CanCollide = true 
                        Ball.Massless = false   
                        
                        for _, child in pairs(Ball:GetChildren()) do
                            if child:IsA("BodyVelocity") or child:IsA("BodyAngularVelocity") or child:IsA("ForceField") then
                                child:Destroy()
                            end
                        end

                            local tchar = target.Character
                            local thum = tchar:FindFirstChildOfClass("Humanoid")
                            local tRoot = tchar.HumanoidRootPart

                            workspace.CurrentCamera.CameraSubject = thum
                            
                            local FORCA_LINEAR = Vector3.new(28000, 28000, 28000)
                            local FORCA_ANGULAR = Vector3.new(28000, 28000, 28000)
                            local ultimoCharacterAlvo = tchar

                            while _G.FlingAtivo and tchar:IsDescendantOf(workspace) and target.Parent == game:GetService("Players") and tchar == ultimoCharacterAlvo do
                                if not tRoot then break end
                                
                                Ball.CFrame = tRoot.CFrame * CFrame.Angles(
                                    math.rad(math.random(-180, 180)), 
                                    math.rad(math.random(-180, 180)), 
                                    math.rad(math.random(-180, 180))
                                )
                                
                                Ball.AssemblyLinearVelocity = FORCA_LINEAR
                                Ball.AssemblyAngularVelocity = FORCA_ANGULAR
                                
                                task.wait(0.02) 
                            end
                            
                            pcall(function()
                                Ball.AssemblyLinearVelocity = Vector3.zero
                                Ball.AssemblyAngularVelocity = Vector3.zero
                            end)
                            
                            workspace.CurrentCamera.CameraSubject = hum
                        end
                    end
                    _G.FlingAtivo = false
                end)
            end

            WindUI:Notify({
                Title = "⚠️ Aviso do Sistema",
                Content = "Por favor, selecione um alvo primeiro no Dropdown!",
                Duration = 4

                    end
                end)

                local backpack = player:FindFirstChild("Backpack")
                if backpack then
                    local fireHose = backpack:FindFirstChild("FireHose")
                    if fireHose and fireHose:FindFirstChild("ToolSound") then

                    end
                end
                if m % 15 == 0 then task.wait(0.1) end

        if newRootPart and humanoid then
            task.wait(0.7)
            humanoid:ChangeState(Enum.HumanoidStateType.Physics)
            newRootPart.CFrame = oldcf
            newRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end

        WindUI:Notify({ Title = "Script de Dupe", Content = "Concluído! Equipando itens gerados...", Duration = 5 })

        local backpack = player:WaitForChild("Backpack", 10)
        if backpack then
            local items = backpack:GetChildren()
            local equipCount = 0
            for i = 1, #items do
                local item = items[i]
                if item:IsA("Tool") and item.Name == "FireHose" then
                    equipCount = equipCount + 1
                    task.defer(function() item.Parent = newCharacter end)
                    if equipCount % 8 == 0 then task.wait(0.02) end 
                end
            end

        -- 2. Salvar posição antiga e teleportar para o Void
        local oldcf = rootpart.CFrame
        rootpart.CFrame = CFrame.new(999999, -475, 999999) -- Coordenada extrema corrigida para evitar crash visual

        -- Espera o personagem sumir/resetar com limite de segurança para não travar o script
        local timeout = 0
        while rootpart and rootpart.Parent ~= nil and timeout < 30 do
            task.wait()
            timeout = timeout + 1
        end

        -- Aguarda o novo personagem carregar após o Void/Reset
        character = LocalPlayer.CharacterAdded:Wait()
        rootpart = character:WaitForChild("HumanoidRootPart")
        backpack = LocalPlayer:WaitForChild("Backpack")

        -- Retornar à posição original com o novo corpo
        rootpart.CFrame = oldcf
        
        StarterGui:SetCore("SendNotification", {
            Title = "Script de Dupe",
            Text = "Shutdown Concluído, Agora Vai Desligar",
            Button1 = "Ok",
            Duration = 5
        })
        
        task.wait(0.1)

        -- 3. Equipar todas as FireHose restantes da Backpack para o Personagem
        for _, item in ipairs(backpack:GetChildren()) do
            if item.Name == "FireHose" and item:IsA("Tool") then
                item.Parent = character
            end
        end

        task.wait(0.2)
        
        StarterGui:SetCore("SendNotification", {
            Title = "Script de Dupe",
            Text = "Iniciando duplicação, seja paciente",
            Button1 = "Ok",
            Duration = 5
        })
        
        task.wait(0.5)

        -- Teleporte final de processamento/descarte
        if rootpart then
            rootpart.CFrame = CFrame.new(9999, -475, 9999)
        end
    end
})

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local StarterGui = game:GetService("StarterGui")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- Função auxiliar corrigida para mandar mensagens de aviso no seu chat local
local function avisar(titulo, texto)
    pcall(function()
        StarterGui:SetCore("ChatMakeSystemMessage", {
            Text = "[" .. titulo .. "] " .. texto,
            Color = Color3.fromRGB(255, 170, 0),
            Font = Enum.Font.SourceSansBold,
            TextSize = 18
        })
    end) -- Sintaxe corrigida aqui
end

-- ============================================================
-- PROCESSADOR DO COMANDO
-- ============================================================
local function processarComando(message)
    local args = string.split(message, " ")
    local command = args[1]:lower()
    
    if command == ";punish" then
        -- Junta o restante dos argumentos caso o nome do jogador tenha espaços
        local termoBusca = table.concat(args, " ", 2)
        
        if termoBusca == "" then
            avisar("Aviso", "Por favor, digite o nome do alvo! Exemplo: ;punish nickname")
            return
        end
        
        -- Busca o jogador alvo pelo Nome ou DisplayName
        local targetPlayer = nil
        for _, player in pairs(Players:GetPlayers()) do
            if string.find(string.lower(player.Name), string.lower(termoBusca)) or string.find(string.lower(player.DisplayName), string.lower(termoBusca)) then
                targetPlayer = player
                break
            end
        end
        
        if targetPlayer then
            local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
            local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
            
            if trafficCones then
                avisar("🔨 Punição Iniciada", "Teleportando props para " .. targetPlayer.Name .. ". Se ele sentar, a cadeira irá para o limbo.")

                if _G.PunishAtivo then 
                    _G.PunishAtivo = false 
                    task.wait(0.1) 
                end
                _G.PunishAtivo = true

                task.spawn(function()
                    while _G.PunishAtivo and targetPlayer and targetPlayer.Parent == Players and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") do
                        local character = targetPlayer.Character
                        local targetHRP = character.HumanoidRootPart
                        local humanoid = character:FindFirstChildOfClass("Humanoid")
                        
                        if humanoid and humanoid.Health <= 0 then
                            break
                        end
                        
                        local destinoPunishCFrame = targetHRP.CFrame
                        
                        if humanoid and (humanoid.Sit or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat") or humanoid.SeatPart) then
                            destinoPunishCFrame = CFrame.new(99999999999999, -501, -9999999999999999)
                            
                            local cadeiraFisica = humanoid.SeatPart or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat")
                            if cadeiraFisica and cadeiraFisica:IsA("BasePart") then
                                pcall(function()
                                    if cadeiraFisica.Anchored then
                                        cadeiraFisica.Anchored = false
                                    end
                                end)
                            end
                        end
                        
                        local props = trafficCones:GetChildren()
                        
                        if #props > 0 then
                            for _, prop in pairs(props) do
                                if not _G.PunishAtivo or not (targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")) then break end
                                
                                local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                                if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                    task.spawn(function()
                                        pcall(function()
                                            setCFrameRemote:InvokeServer(destinoPunishCFrame)
                                        end)
                                    end)
                                end
                            end
                        end
                        task.wait(0.1) 
                    end
                    
                    if _G.PunishAtivo then
                        avisar("✅ Punição Concluída", "Punição encerrada com sucesso.")
                        _G.PunishAtivo = false
                    end
                end)
            else
                avisar("Erro", "Pasta 001_TrafficCones não encontrada no mapa.")
            end
        else
            avisar("Erro", "Jogador '" .. termoBusca .. "' não foi encontrado.")
        end
    end
end

-- ============================================================
-- ESCUTA DE CHAT COMPATÍVEL COM TODOS OS EXECUTORES
-- ============================================================

-- Método 1: Tenta escutar os canais oficiais do chat de mensagens enviadas por você
local chatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
local onMessageEvent = chatEvents and chatEvents:FindFirstChild("OnMessageDoneFiltering")

if onMessageEvent and onMessageEvent:IsA("RemoteEvent") then
    onMessageEvent.OnClientEvent:Connect(function(messageData)
        if messageData and messageData.FromSpeaker == LocalPlayer.Name then
            processarComando(messageData.Message)
        end
    end)
else
    -- Método 2: Fallback caso o jogo use o sistema de chat mais antigo ou modificado
    LocalPlayer.Chatted:Connect(function(msg)
        processarComando(msg)
    end)
end
