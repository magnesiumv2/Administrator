-- ============================================================
-- 1. SISTEMA DO DROPDOWN AUTOMÁTICO (WINDUI)
-- ==========================================================
local AlvoSelecionado = ""

-- Função modificada para obter a lista de jogadores atuais
local function obterListaJogadores()
    local lista = {}
    for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
        if p ~= game:GetService("Players").LocalPlayer then
            table.insert(lista, p.Name)
        end
    end
    return lista
end

-- ============================================================
-- 1. SELEÇÃO DE ALVO (Dropdown Nativo da WindUI)
-- ============================================================
local DropdownAlvos = MainTab:Dropdown({
    Title = "Selecionar Alvo",
    Desc = "Escolha um jogador do servidor para executar ações",
    Values = obterListaJogadores(),
    Value = "", -- Começa sem nenhum alvo pré-definido
    Callback = function(OpcaoSelecionada)
        -- Normaliza o retorno caso a interface retorne uma tabela de seleção múltipla
        if type(OpcaoSelecionada) == "table" then
            AlvoSelecionado = OpcaoSelecionada[1] or ""
        else
            AlvoSelecionado = OpcaoSelecionada or ""
        end
        print("Alvo atual definido para: " .. tostring(AlvoSelecionado))
    end
})

-- Função automatizada para sincronizar a lista interna com o componente visual
local function atualizarDropdown()
    local novaLista = obterListaJogadores()
    DropdownAlvos:Refresh(novaLista) -- Método nativo suportado pela WindUI
end

-- Conexões de eventos para atualizar a lista dinamicamente
game:GetService("Players").PlayerAdded:Connect(function()
    task.wait(0.5)
    atualizarDropdown()
end)

game:GetService("Players").PlayerRemoving:Connect(function(player)
    if AlvoSelecionado == player.Name then
        AlvoSelecionado = ""
    end
    atualizarDropdown()
end)

-- 2. BOTÃO: ;fling (Adaptado)
