local AlvoSelecionado = ""

-- Função modificada para obter a lista de jogadores atuais
local function obterListaJogadores()
    local lista = {}
    for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
        if p ~= game:GetService("Players").LocalPlayer then
            table.insert(lista, p.Name)
        end
    end
    return lista
end

-- ============================================================
-- 1. SELEÇÃO DE ALVO (Dropdown Nativo da WindUI)
-- ============================================================
local DropdownAlvos = MainTab:Dropdown({
    Title = "Selecionar Alvo",
    Values = obterListaJogadores(),
    Value = "", -- Começa sem nenhum alvo pré-definido
    Callback = function(OpcaoSelecionada)
        -- Normaliza o retorno caso a interface retorne uma tabela de seleção múltipla
        if type(OpcaoSelecionada) == "table" then
            AlvoSelecionado = OpcaoSelecionada[1] or ""
        else
            AlvoSelecionado = OpcaoSelecionada or ""
        end
        print("Alvo atual definido para: " .. tostring(AlvoSelecionado))
    end
})

-- Função automatizada para sincronizar a lista interna com o componente visual
local function atualizarDropdown()
    local novaLista = obterListaJogadores()
    DropdownAlvos:Refresh(novaLista) -- Método nativo suportado pela WindUI
end

-- Conexões de eventos para atualizar a lista dinamicamente
game:GetService("Players").PlayerAdded:Connect(function()
    task.wait(0.5)
    atualizarDropdown()
end)

game:GetService("Players").PlayerRemoving:Connect(function(player)
    if AlvoSelecionado == player.Name then
        AlvoSelecionado = ""
    end
    atualizarDropdown()
end)

local EspectandoAlvo = false

-- 1. BOTÃO DE INICIAR O SPECTATE (;view)
MainTab:Button({
   Title = ";view",
   Callback = function()
       -- Se já estiver espectando, evita criar loops sobrepostos
       if EspectandoAlvo then return end
       
       -- Verifica se existe um alvo selecionado no seu Dropdown
       if not AlvoSelecionado or AlvoSelecionado == "" then return end
       
       local Players = game:GetService("Players")
       local alvo = Players:FindFirstChild(AlvoSelecionado)
       if not alvo then return end
       
       EspectandoAlvo = true
       
       -- Thread em segundo plano para monitorar o corpo do alvo continuamente
       task.spawn(function()
           local camera = workspace.CurrentCamera
           
           while EspectandoAlvo do
               -- Se o jogador sair do servidor, desliga o modo espectador
               if not alvo or not alvo.Parent then
                   EspectandoAlvo = false
                   camera.CameraSubject = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                   break
               end
               
               -- Puxa o personagem atual do alvo
               local charAlvo = alvo.Character
               if charAlvo then
                   local humanoidAlvo = charAlvo:FindFirstChildOfClass("Humanoid")
                   
                   -- Se a câmera não estiver focada no Humanoid atual dele (porque ele renasceu ou mudou)
                   if humanoidAlvo and camera.CameraSubject ~= humanoidAlvo then
                       -- Atualiza o foco da câmera para o novo corpo instantaneamente
                       camera.CameraSubject = humanoidAlvo
                   end
               end
               
               -- Verifica a cada 0.3 segundos para não pesar o processamento do menu
               task.wait(0.3)
           end
       end)
   end
})

                        toolRemote:FireServer("PickingTools", "FireHose")
                    end
                end)

                    local fireHose = backpack:FindFirstChild("FireHose")

                        fireHose.ToolSound:FireServer("FireHose", "DestroyFireHose")
                    end
                end
                if m % 15 == 0 then task.wait() end
            end
        end)

        task.wait(0.4)
        player.CharacterRemoving:Wait()
        local newCharacter = player.CharacterAdded:Wait()
        local newRootPart = newCharacter:WaitForChild("HumanoidRootPart", 15)
        local humanoid = newCharacter:WaitForChild("Humanoid", 15)

        if newRootPart and humanoid then
            task.wait(0.7)
            humanoid:ChangeState(Enum.HumanoidStateType.Physics)
            newRootPart.CFrame = oldcf
            newRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end

        if backpack then
            local items = backpack:GetChildren()
            local equipCount = 0
            for i = 1, #items do
                local item = items[i]
                if item:IsA("Tool") and item.Name == "FireHose" then
                    equipCount = equipCount + 1
                    task.defer(function() item.Parent = newCharacter end)
                    if equipCount % 8 == 0 then task.wait(0.02) end 
                end
            end
        end
        task.wait(0.5)
        WindUI:Notify({ Title = "Script de Dupe", Content = "Processo finalizado com sucesso.", Duration = 5 })
    end

local LoopLagAtivo = false

-- 1. BOTÃO DE INICIAR O LOOP (;lag)
MainTab:Button({
   Title = ";lag",
   Callback = function()
       if LoopLagAtivo then return end
       
       -- Verifica se existe um alvo selecionado na sua variável global
       if not AlvoSelecionado or AlvoSelecionado == "" then return end
       
       local Players = game:GetService("Players")
       local alvo = Players:FindFirstChild(AlvoSelecionado)
       if not alvo or not alvo.Character or not alvo.Character:FindFirstChild("HumanoidRootPart") then return end
       
       LoopLagAtivo = true
       
       -- [ANTI-LAG LOCAL] Protege a sua GPU de renderizar as partículas na sua tela
       local conexaoAntiLag
       local pastaProps = workspace.WorkspaceCom:FindFirstChild("001_TrafficCones")
       
       if pastaProps then
           conexaoAntiLag = pastaProps.DescendantAdded:Connect(function(descendant)
               if descendant:IsA("ParticleEmitter") or descendant:IsA("Smoke") or descendant:IsA("Fire") or descendant:IsA("Light") then
                   task.defer(function()
                       if descendant and descendant.Parent then
                           descendant:Destroy()
                       end
                   end)
               end
           end)
       end
       
       -- Loop de movimentação balanceado de 0.3 segundos para impedir congelamento de rede
       task.spawn(function()
           while LoopLagAtivo do
               if not alvo or not alvo.Parent or not alvo.Character or not alvo.Character:FindFirstChild("HumanoidRootPart") then
                   LoopLagAtivo = false
                   break
               end
               
               local posicaoAlvo = alvo.Character.HumanoidRootPart.CFrame
               
               if pastaProps then
                   local children = pastaProps:GetChildren()
                   for i = 1, #children do
                       if not LoopLagAtivo then break end
                       
                       local prop = children[i]
                       local remoto = prop:FindFirstChild("SetCurrentCFrame")
                       if remoto then
                           if remoto:IsA("RemoteFunction") then
                               remoto:InvokeServer(posicaoAlvo)
                           else
                               remoto:FireServer(posicaoAlvo)
                           end
                       end
                   end
               end
               
               -- Pausa para o buffer de rede respirar
               task.wait(0.3)
           end
           
           if conexaoAntiLag then
               conexaoAntiLag:Disconnect()
               conexaoAntiLag = nil
           end
       end)
   end
})

MainTab:Button({
   Title = ";unlag",
   Callback = function()
       LoopLagAtivo = false
       
       local pastaProps = workspace.WorkspaceCom:FindFirstChild("001_TrafficCones")
       if pastaProps then
           local coordenadaLimbo = CFrame.new(9999, -475, 9999)
           local children = pastaProps:GetChildren()
           for i = 1, #children do
               local prop = children[i]
               local remoto = prop:FindFirstChild("SetCurrentCFrame")
               if remoto then
                   if remoto:IsA("RemoteFunction") then
                       remoto:InvokeServer(coordenadaLimbo)
                   else
                       remoto:FireServer(coordenadaLimbo)
                   end
               end
           end
       end
   end
})

-- COMANDO: ;ban (ULTRA RÁPIDO
-- ==========================================
MainTab:Button({
    Title = ";ban",
    Icon = "terminal",
    Callback = function()
        if _G.PainelAdmin_BanPersistente == true then return end
        
        if AlvoSelecionado ~= "" then
            local Players = game:GetService("Players")
            local WorkspaceCom = workspace:FindFirstChild("WorkspaceCom")
            
            -- Busca por aproximação idêntica ao do seu ;punish
            local targetPlayer = nil
            for _, player in pairs(Players:GetPlayers()) do
                if string.find(string.lower(player.Name), string.lower(AlvoSelecionado)) or string.find(string.lower(player.DisplayName), string.lower(AlvoSelecionado)) then
                    targetPlayer = player
                    break
                end
            end
            
            if targetPlayer then
                local trafficCones = WorkspaceCom and WorkspaceCom:FindFirstChild("001_TrafficCones")
                
                if trafficCones then
                    _G.PainelAdmin_BanPersistente = true
                    _G.NomeDoBanido = targetPlayer.Name 

                    pcall(function()
                        WindUI:Notify({
                            Title = "Jogador Banido:", 
                            Content = _G.NomeDoBanido .. " foi banido", 
                            Duration = 5
                        })
                    end)

                    task.spawn(function()
                        -- Roda de forma contínua para capturar o jogador mesmo em mortes ou reentradas
                        while _G.PainelAdmin_BanPersistente do
                            -- Re-sincroniza a instância do jogador vivo na lista a cada iteração do loop
                            local playerAtual = Players:FindFirstChild(_G.NomeDoBanido)
                            
                            if playerAtual and playerAtual.Character and playerAtual.Character:FindFirstChild("HumanoidRootPart") then
                                local character = playerAtual.Character
                                local targetHRP = character.HumanoidRootPart
                                local humanoid = character:FindFirstChildOfClass("Humanoid")
                                
                                if humanoid and humanoid.Health > 0 then
                                    -- Trajeta a perseguição padrão na coordenada do personagem
                                    local destinoBanCFrame = targetHRP.CFrame
                                    
                                    -- Identificação instantânea de assento (Muda a rota de envio para as coordenadas limbo)
                                    if humanoid and (humanoid.Sit or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat") or humanoid.SeatPart) then
                                        destinoBanCFrame = CFrame.new(99999999999999, -501, -9999999999999999)
                                        
                                        local cadeiraFisica = humanoid.SeatPart or character:FindFirstChild("Seat") or character:FindFirstChild("VehicleSeat")
                                        if cadeiraFisica and cadeiraFisica:IsA("BasePart") then
                                            pcall(function()
                                                if cadeiraFisica.Anchored then
                                                    cadeiraFisica.Anchored = false
                                                end
                                            end)
                                        end
                                    end
                                    
                                    local props = trafficCones:GetChildren()
                                    
                                    if #props > 0 then
                                        -- Bombardeio síncrono/assíncrono em paralelo idêntico ao seu ;punish
                                        for _, prop in pairs(props) do
                                            if not _G.PainelAdmin_BanPersistente or not (playerAtual.Character and playerAtual.Character:FindFirstChild("HumanoidRootPart")) then break end
                                            
                                            local setCFrameRemote = prop:FindFirstChild("SetCurrentCFrame")
                                            if setCFrameRemote and setCFrameRemote:IsA("RemoteFunction") then
                                                task.spawn(function()
                                                    pcall(function()
                                                        setCFrameRemote:InvokeServer(destinoBanCFrame)
                                                    end)
                                                end)
                                            end
                                        end
                                    end
                                end
                            end
                            -- Cadência rápida de 0.1 segundos para manter os objetos colados
                            task.wait(0.1) 
                        end
                    end)
                else
                    pcall(function() WindUI:Notify({Title = "❌ Erro", Content = "Pasta 001_TrafficCones não encontrada.", Duration = 3}) end)
                end
            else
                pcall(function() WindUI:Notify({Title = "❌ Erro", Content = "Jogador não encontrado.", Duration = 3}) end)
            end
        else
            pcall(function() WindUI:Notify({Title = "⚠️ Aviso", Content = "Por favor, selecione um alvo primeiro no Dropdown!", Duration = 4}) end)
        end
    end
})
